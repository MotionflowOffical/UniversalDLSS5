if(WIN32)
  include(cmake/Dependencies.cmake)

  add_library(udlss_profile STATIC src/common/profile.cpp src/common/shared_control_win.cpp)
  target_link_libraries(udlss_profile PUBLIC udlss_common advapi32)

  add_executable(UniversalDLSS5 WIN32
    src/controller/main.cpp src/controller/processes.cpp src/controller/inject_client.cpp src/controller/ui.cpp)
  target_compile_definitions(UniversalDLSS5 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(UniversalDLSS5 PRIVATE udlss_profile comctl32 shell32 shlwapi version)

  add_executable(UniversalDLSS5.Injector src/injector/main.cpp src/injector/injector.cpp)
  target_link_libraries(UniversalDLSS5.Injector PRIVATE udlss_common advapi32)

  add_library(UniversalDLSS5.Bridge SHARED
    src/bridge/bridge_main.cpp src/bridge/dxgi_hooks.cpp
    src/gpu/d3d11_pipeline.cpp src/gpu/d3d11_guide_extractor.cpp src/gpu/d3d11_resource_tracker.cpp src/gpu/d3d11_camera_tracker.cpp src/gpu/d3d12_on12.cpp src/gpu/nv_optical_flow.cpp
    src/neural/passthrough.cpp src/neural/ingame_nr.cpp src/neural/ngx_nr.cpp src/neural/external_host.cpp src/neural/streamline_nr.cpp
    src/common/shared_control_win.cpp)
  target_compile_definitions(UniversalDLSS5.Bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(UniversalDLSS5.Bridge PRIVATE
    udlss_common minhook d3d11 d3d12 dxgi d3dcompiler dxguid shlwapi advapi32)
  if(UDLSS_WITH_STREAMLINE)
    target_compile_definitions(UniversalDLSS5.Bridge PRIVATE UDLSS_WITH_STREAMLINE=1)
    target_include_directories(UniversalDLSS5.Bridge PRIVATE ${UDLSS_STREAMLINE_ROOT}/include)
  endif()

  if(UDLSS_WITH_NGX_NR AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(UDLSS_NGX_LIB "${UDLSS_NGX_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_d.lib")
    if(NOT EXISTS "${UDLSS_NGX_ROOT}/include/nvsdk_ngx.h" OR NOT EXISTS "${UDLSS_NGX_LIB}")
      message(FATAL_ERROR "NVIDIA NGX SDK checkout is incomplete at ${UDLSS_NGX_ROOT}")
    endif()
    message(STATUS "NVIDIA NGX DLSS-NR backend: ${UDLSS_NGX_ROOT}")
    target_compile_definitions(UniversalDLSS5.Bridge PRIVATE UDLSS_WITH_NGX_NR=1)
    target_include_directories(UniversalDLSS5.Bridge PRIVATE "${UDLSS_NGX_ROOT}/include")
    target_link_libraries(UniversalDLSS5.Bridge PRIVATE "${UDLSS_NGX_LIB}")
    add_executable(UniversalDLSS5.NRHost WIN32 src/host/main.cpp)
    target_compile_definitions(UniversalDLSS5.NRHost PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(UniversalDLSS5.NRHost PRIVATE "${UDLSS_NGX_ROOT}/include" ${CMAKE_CURRENT_SOURCE_DIR}/src/neural)
    target_link_libraries(UniversalDLSS5.NRHost PRIVATE udlss_common minhook d3d12 dxgi shell32 "${UDLSS_NGX_LIB}")

    add_library(UniversalDLSS5.NRForwarder SHARED src/host/nr_forwarder.cpp)
    target_compile_definitions(UniversalDLSS5.NRForwarder PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(UniversalDLSS5.NRForwarder PRIVATE "${UDLSS_NGX_ROOT}/include")
    target_link_libraries(UniversalDLSS5.NRForwarder PRIVATE d3d12)
    set_target_properties(UniversalDLSS5.NRForwarder PROPERTIES
      PREFIX ""
      OUTPUT_NAME "nvngx.dll_UniversalDLSS5_NRForwarder")
  elseif(UDLSS_WITH_NGX_NR)
    message(STATUS "NVIDIA NGX DLSS-NR backend disabled for 32-bit bridge; passthrough remains available")
  endif()

  if(UDLSS_WITH_NVOF)
    find_path(UDLSS_NVOF_INCLUDE_DIR nvOpticalFlowD3D11.h
      HINTS
        "${UDLSS_NVOF_ROOT}/NvOFInterface"
        "${UDLSS_NVOF_ROOT}/include"
        "$ENV{NVOF_SDK_ROOT}/NvOFInterface"
        "$ENV{NVOF_SDK_ROOT}/include")
    if(UDLSS_NVOF_INCLUDE_DIR)
      message(STATUS "NVIDIA Optical Flow headers: ${UDLSS_NVOF_INCLUDE_DIR}")
      target_compile_definitions(UniversalDLSS5.Bridge PRIVATE UDLSS_WITH_NVOF=1)
      target_include_directories(UniversalDLSS5.Bridge PRIVATE "${UDLSS_NVOF_INCLUDE_DIR}")
    else()
      message(STATUS "NVIDIA Optical Flow headers not found; HLSL optical-flow fallback remains enabled")
    endif()
  endif()

  add_executable(UniversalDLSS5.TestApp WIN32 src/testapp/main.cpp)
  target_link_libraries(UniversalDLSS5.TestApp PRIVATE d3d11 dxgi d3dcompiler)

  add_custom_command(TARGET UniversalDLSS5.Bridge POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:UniversalDLSS5.Bridge>/shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:UniversalDLSS5.Bridge>/shaders"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:UniversalDLSS5.Bridge>/runtime"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/runtime" "$<TARGET_FILE_DIR:UniversalDLSS5.Bridge>/runtime")

  set(_udlss_runtime_targets UniversalDLSS5 UniversalDLSS5.Injector UniversalDLSS5.Bridge UniversalDLSS5.TestApp)
  if(TARGET UniversalDLSS5.NRHost)
    list(APPEND _udlss_runtime_targets UniversalDLSS5.NRHost)
  endif()
  if(TARGET UniversalDLSS5.NRForwarder)
    list(APPEND _udlss_runtime_targets UniversalDLSS5.NRForwarder)
  endif()
  set_target_properties(${_udlss_runtime_targets}
    PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
